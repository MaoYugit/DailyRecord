### 第一部分：一键建库脚本 (SQL)

请在 MySQL 客户端（如 Navicat / DBeaver）中运行以下代码：

```sql
-- 1. 创建数据库 (使用 utf8mb4 以支持 Emoji 表情)
CREATE DATABASE IF NOT EXISTS blog_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE blog_db;

-- 2. 用户表 (users)
-- 存储博主和注册用户
CREATE TABLE `users` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `username` varchar(60) NOT NULL COMMENT '用户名',
  `password_hash` varchar(255) NOT NULL COMMENT '哈希密码(BCrypt)',
  `email` varchar(100) NOT NULL COMMENT '邮箱',
  `nickname` varchar(60) DEFAULT NULL COMMENT '显示昵称',
  `avatar` varchar(255) DEFAULT NULL COMMENT '头像URL',
  `bio` varchar(255) DEFAULT NULL COMMENT '简介',
  `role` enum('admin','editor','subscriber') NOT NULL DEFAULT 'subscriber' COMMENT '角色权限',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_username` (`username`),
  UNIQUE KEY `uk_email` (`email`)
) ENGINE=InnoDB COMMENT='用户表';

-- 3. 分类表 (categories)
-- 文章的目录结构，支持多级
CREATE TABLE `categories` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `name` varchar(60) NOT NULL COMMENT '分类名称',
  `slug` varchar(60) NOT NULL COMMENT 'SEO友好URL',
  `description` varchar(200) DEFAULT NULL,
  `parent_id` bigint(20) unsigned DEFAULT '0' COMMENT '父分类ID(0为顶级)',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_slug` (`slug`),
  KEY `idx_parent_id` (`parent_id`)
) ENGINE=InnoDB COMMENT='分类表';

-- 4. 标签表 (tags)
-- 文章的关键词，扁平结构
CREATE TABLE `tags` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `name` varchar(60) NOT NULL,
  `slug` varchar(60) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_slug` (`slug`)
) ENGINE=InnoDB COMMENT='标签表';

-- 5. 文章主表 (posts)
-- 核心内容表
CREATE TABLE `posts` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `author_id` bigint(20) unsigned NOT NULL COMMENT '作者ID',
  `category_id` bigint(20) unsigned DEFAULT NULL COMMENT '主分类ID',
  `title` varchar(255) NOT NULL COMMENT '标题',
  `slug` varchar(255) NOT NULL COMMENT 'URL别名',
  `excerpt` varchar(500) DEFAULT NULL COMMENT '摘要',
  `content` longtext NOT NULL COMMENT 'Markdown源码',
  `content_html` longtext DEFAULT NULL COMMENT '渲染后的HTML(缓存)',
  `status` enum('published','draft','archived') NOT NULL DEFAULT 'draft' COMMENT '状态',
  `type` enum('post','page') NOT NULL DEFAULT 'post' COMMENT '类型:文章/页面',
  `view_count` int(10) unsigned DEFAULT '0' COMMENT '基础阅读量',
  `comment_count` int(10) unsigned DEFAULT '0' COMMENT '评论数缓存',
  `published_at` datetime DEFAULT NULL COMMENT '发布时间',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_slug` (`slug`),
  KEY `idx_author` (`author_id`),
  KEY `idx_category` (`category_id`),
  KEY `idx_status_date` (`status`, `published_at`), -- 常用查询优化索引
  CONSTRAINT `fk_posts_users` FOREIGN KEY (`author_id`) REFERENCES `users` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_posts_categories` FOREIGN KEY (`category_id`) REFERENCES `categories` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB COMMENT='文章表';

-- 6. 文章-标签关联表 (post_tags)
-- 解决多对多关系
CREATE TABLE `post_tags` (
  `post_id` bigint(20) unsigned NOT NULL,
  `tag_id` bigint(20) unsigned NOT NULL,
  PRIMARY KEY (`post_id`,`tag_id`),
  KEY `idx_tag_id` (`tag_id`),
  CONSTRAINT `fk_pt_posts` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_pt_tags` FOREIGN KEY (`tag_id`) REFERENCES `tags` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='文章标签关联';

-- 7. 文章扩展属性表 (post_meta)
-- ★★★ 核心扩展表：存储自定义字段 ★★★
CREATE TABLE `post_meta` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `post_id` bigint(20) unsigned NOT NULL,
  `meta_key` varchar(255) NOT NULL COMMENT '键(如 banner_img)',
  `meta_value` longtext COMMENT '值',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_post_meta` (`post_id`, `meta_key`), -- 确保一篇文章的同一个key唯一
  CONSTRAINT `fk_meta_posts` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='文章扩展属性';

-- 8. 评论表 (comments)
-- 支持嵌套回复
CREATE TABLE `comments` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `post_id` bigint(20) unsigned NOT NULL,
  `user_id` bigint(20) unsigned DEFAULT NULL COMMENT '登录用户ID(空为游客)',
  `parent_id` bigint(20) unsigned DEFAULT '0' COMMENT '父评论ID',
  `author_name` varchar(60) NOT NULL COMMENT '显示名称',
  `author_email` varchar(100) NOT NULL COMMENT '邮箱(用于Gravatar)',
  `author_website` varchar(200) DEFAULT NULL,
  `content` text NOT NULL,
  `is_admin` tinyint(1) NOT NULL DEFAULT '0' COMMENT '是否博主回复',
  `status` enum('approved','pending','spam') NOT NULL DEFAULT 'pending',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_post_id` (`post_id`),
  KEY `idx_parent_id` (`parent_id`),
  CONSTRAINT `fk_comments_posts` FOREIGN KEY (`post_id`) REFERENCES `posts` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='评论表';

-- 9. 资源附件表 (attachments)
-- 媒体库管理
CREATE TABLE `attachments` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint(20) unsigned NOT NULL,
  `original_name` varchar(255) NOT NULL,
  `file_path` varchar(255) NOT NULL COMMENT '相对路径',
  `file_url` varchar(255) NOT NULL COMMENT '完整URL',
  `file_type` varchar(50) NOT NULL COMMENT 'MIME类型',
  `file_size` int(10) unsigned NOT NULL,
  `storage_driver` varchar(20) NOT NULL DEFAULT 'local' COMMENT '存储驱动:local,oss,s3',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_user` (`user_id`)
) ENGINE=InnoDB COMMENT='附件表';

-- 10. 系统配置表 (options)
-- 全局键值对设置
CREATE TABLE `options` (
  `id` int(10) unsigned NOT NULL AUTO_INCREMENT,
  `option_key` varchar(100) NOT NULL,
  `option_value` longtext,
  `autoload` tinyint(1) NOT NULL DEFAULT '1' COMMENT '初始化时是否加载',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_option_key` (`option_key`)
) ENGINE=InnoDB COMMENT='配置表';

-- 初始化基础配置
INSERT INTO `options` (`option_key`, `option_value`) VALUES 
('site_title', 'My Awesome Blog'),
('site_url', 'http://localhost:8080'),
('posts_per_page', '10');
```

### 第二部分：深度解析（为什么这么设计？）

作为开发者，理解设计意图比死记代码更重要。下面是每张表的设计核心逻辑：

#### 1. 基础设计规范
*   **`utf8mb4`**: 现在是2023+年了，必须支持 Emoji 表情（😀）。传统的 `utf8` 在 MySQL 中只支持 3 字节，存不下表情，`utf8mb4` 才是真正的 UTF-8。
*   **`bigint(20) unsigned`**: 主键 ID 使用无符号大整数。博客运行久了，ID 可能很大，`unsigned` 保证没有负数，容量翻倍。
*   **`created_at` / `updated_at`**: 几乎每张表都有。对于数据审计和“最近更新”功能至关重要。

#### 2. 用户表 (`users`)
*   **`password_hash`**: **永远不要存明文密码！** 即使是个人博客。这里预留 255 长度，足够存放 BCrypt 或 Argon2 加密后的字符串。
*   **`role`**: 使用 ENUM 枚举。这比单独搞一个 `roles` 表更简单高效，适合博客这种角色固定的系统（管理员、编辑、普通读者）。

#### 3. 文章主表 (`posts`) —— 系统的灵魂
*   **`slug` (URL别名)**:
    *   *Bad*: `blog.com/post?id=105` (对搜索引擎不友好，看不出内容)
    *   *Good*: `blog.com/post/learn-mysql-design` (SEO 友好)
    *   设计中要求 `UNIQUE`，保证链接唯一。
*   **`content` vs `content_html`**:
    *   `content` 存你写的 Markdown 源码。
    *   `content_html` 存解析后的 HTML。
    *   *为什么要两个？* 前端直接读 `content_html` 速度极快，不用每次请求都由后端解析 Markdown；当你需要“编辑”文章时，调出 `content` 源码。
*   **`status` & `published_at`**:
    *   将发布时间与创建时间分开。你可以今天写好文章（`created_at`），设定为后天发布（`published_at` = 后天）。后端查询时只查 `published_at <= NOW()` 的文章。

#### 4. 扩展属性表 (`post_meta`) —— 赋予你无限能力
这是最关键的“第9张表”。
*   **场景**: 你想给某一篇文章加一个“背景音乐”或者“推荐指数”。
*   **做法**: 不要去改 `posts` 表结构！直接在 `post_meta` 插入一行：
    *   `post_id`: 10, `meta_key`: 'bgm_url', `meta_value`: 'music.mp3'
*   **优势**: WordPress 靠这个机制实现了成千上万的插件功能。你的博客因此具备了扩展性。

#### 5. 分类与标签 (`categories` & `tags`)
*   **区别**:
    *   **分类 (`categories`)**: 有 `parent_id`，它是树状的（如：技术 -> 后端 -> Java）。一篇文章通常只属于一个主分类。
    *   **标签 (`tags`)**: 扁平的。一篇文章可以贴很多标签（如：#MySQL #数据库 #教程）。
*   **中间表 (`post_tags`)**: 因为是多对多关系，必须用中间表。注意 `ON DELETE CASCADE`，这意味着如果你删除了文章，关联关系会自动删除，不会留下垃圾数据。

#### 6. 评论表 (`comments`)
*   **混合模式**: `user_id` 允许为 NULL。
    *   如果用户登录了，存 ID。
    *   如果用户没登录（游客），存 `author_name` 和 `email`。
*   **`parent_id`**: 实现“楼中楼”回复。顶级评论为 0，回复某人的评论则填入对方的 ID。

#### 7. 资源表 (`attachments`)
*   **Storage Driver**: 字段 `storage_driver`。
    *   刚开始你可能存服务器本地（local）。
    *   后来流量大了，你把图片搬到了阿里云 OSS。
    *   通过这个字段，你的代码可以判断去哪里读取文件，实现平滑迁移。

#### 8. 配置表 (`options`)
*   **Key-Value 存储**:
    *   `site_title`: "我的博客"
    *   `seo_keywords`: "Java, Python, Life"
*   **Autoload**:
    *   设计 `autoload` 字段。系统启动时，把 `autoload=1` 的配置一次性全部读入缓存（Redis 或 内存），避免每打开一个页面都去查几十次数据库。

### 第三部分：实际开发中的 SQL 查询演示

为了让你更有底气，这里演示几个实际开发中会用到的复杂查询：

**1. 查询文章详情 + 分类名 + 作者名**
```sql
SELECT 
    p.title, 
    p.content_html, 
    p.published_at,
    c.name AS category_name,
    u.nickname AS author_name
FROM posts p
LEFT JOIN categories c ON p.category_id = c.id
LEFT JOIN users u ON p.author_id = u.id
WHERE p.slug = 'learn-mysql-design' 
  AND p.status = 'published';
```

**2. 查询某一篇文章的所有标签**
```sql
SELECT t.name, t.slug 
FROM tags t
INNER JOIN post_tags pt ON t.id = pt.tag_id
WHERE pt.post_id = 1;
```

**3. 获取文章的自定义属性（如：获取文章的头图）**
```sql
SELECT meta_value 
FROM post_meta 
WHERE post_id = 1 AND meta_key = 'banner_image';
```

**4. 获取某篇文章的评论树（简单的两层查询思路）**
在后端代码中，通常先查出所有 `post_id = 1` 的评论，然后在代码（Java/Python/PHP）中根据 `parent_id` 组装成树状结构，而不是在 SQL 里做复杂的递归。

### 总结
这套 `blog_db` 设计方案兼顾了**性能**（索引优化）、**扩展性**（Meta表）、**SEO**（Slug）和**数据完整性**（外键约束）。它是一个标准的、可用于生产环境的个人博客数据库架构。你可以直接拿去写代码了！
